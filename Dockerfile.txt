# Deploy the NestJS API and the React UI as ONE Node.js service.
# Explicit Dockerfile prevents the static-site/Caddy auto-detection seen in DockHosting logs.
FROM node:22-alpine
WORKDIR /app

# NestJS TypeScript and Vite both need development dependencies at build time.
COPY package*.json ./
RUN npm install --include=dev --no-audit --no-fund

COPY . .
RUN npm run build

ENV NODE_ENV=production
ENV PORT=3000
EXPOSE 3000

# Important: launch the actual NestJS server, NOT Caddy.
CMD ["npm", "start"]
